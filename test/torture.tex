\ibaudiotext{A}{You are going to hear a conversation recorded at a hydrological
  research station on the upper reaches of the Rhône, concerning sediment.}

% --- options that wrap to two and three lines ------------------------------
\ibrubricmcq
\begin{ibmcq}[B]{According to the hydrologist, what makes the winter
  measurements at the upper gauging station considerably harder to interpret
  than those taken during the summer months?}
  \opt{The instruments are recalibrated on a rolling schedule that does not
       align with the seasonal transition, so readings taken either side of it
       cannot be compared directly without adjustment.}
  \opt{Meltwater arriving from the tributaries carries a suspended load that
       varies far more from hour to hour than it does in summer.}
  \opt{Access to the station is restricted.}
\end{ibmcq}

\begin{ibtrue}[A, C, D, F, I]{5}
  \opt{The station has been recording continuously since 1974, with only two
       interruptions, both caused by damage to the intake structure.}
  \opt{Short.}
  \opt{The sediment budget for the catchment is dominated by a small number of
       high-magnitude events rather than by the everyday background transport
       that the instruments record most of the time.}
  \opt{Bedload and suspended load are measured by different methods.}
  \opt{Another short one.}
  \opt{The revised protocol asks technicians to take duplicate samples at every
       third visit, which has roughly doubled the time each visit takes.}
  \opt{Calibration drift is corrected annually.}
  \opt{Downstream reservoirs trap a proportion of the load.}
  \opt{The published figures understate the total because they omit the
       contribution of the two ungauged tributaries entering below the station,
       which between them drain almost a fifth of the catchment area.}
  \opt{The team publishes its raw data.}
\end{ibtrue}
