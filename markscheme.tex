%% The answer key. It reads the .ans file that the question booklet writes when
%% it compiles, so the key can never drift out of step with the paper.
%%
%%   tectonic -X compile reading-question-booklet.tex   % writes the .ans file
%%   tectonic -X compile markscheme.tex
%%
%% Point \ibmarkschemesource at whichever booklet you want keyed. The booklet
%% records where each text begins and what it was worth, so this file builds
%% one fully ruled table per text, each on its own landscape page with its own
%% total — the layout the IB uses.
%%
%% To add the "Accept" / "Do not accept" columns for a question, declare them
%% before \ibmarkschemetable with \ibnote{<q>}{<accept>}{<do not accept>}.
%% Use \ibbullets{\item …\item …} inside either for the IB's bulleted variants.
%% ---------------------------------------------------------------------------

\newcommand\ibmarkschemetable{%
  % a text heading, then a fresh ruled table
  \def\ibeb@sec##1{%
    \ifibeb@firstsec \global\ibeb@firstsecfalse \else \clearpage \fi
    {\raggedright\noindent\textbf{##1}\par}\vspace{0.17mm}%
    \begin{longtable}{|>{\ibmscell}p{\ibmsQ}|>{\ibmscell}p{\ibmsA}%
                      |>{\ibmscell}p{\ibmsAcc}|>{\ibmscell}p{\ibmsRej}%
                      |>{\centering\arraybackslash\ibmsface}m{\ibmsM}|}
      \hline
      \multicolumn{1}{|c|}{\ibmsstrut\textbf{Q}} &
      \multicolumn{1}{c|}{\textbf{Target answer}} &
      \multicolumn{1}{c|}{\textbf{Accept}} &
      \multicolumn{1}{c|}{\textbf{Do not accept}} &
      \multicolumn{1}{c|}{\textbf{Marks}} \\
      \hline
      \endhead}%
  % one question
  \def\ibeb@row##1##2##3##4{%
    \ibmsstrut\hspace*{0.76mm}\textbf{##1.} & ##2 &
    \ibeb@lookup{acc}{##1} & \ibeb@lookup{rej}{##1} & \textbf{##3} \\
    \hline}%
  % close the table and hang the total box off its bottom right
  \def\ibeb@sub##1{%
    \end{longtable}%
    \par\nointerlineskip\vskip-\arrayrulewidth
    \hbox to \textwidth{\hfil
    \begin{tabular}{|>{\centering\arraybackslash}p{\ibmstotL}%
                    |>{\centering\arraybackslash}p{\ibmstotR}|}
      \hline
      \rule[-\ibmstotd]{0pt}{\ibmstoth}\textbf{Total} & \textbf{##1} \\
      \hline
    \end{tabular}%
    \hspace*{\ibmstotinset}}%
    \par}%
  \ibeb@rows}
\makeatother

% ── marking instructions ───────────────────────────────────────────────────
\begingroup\ibmsface
\begin{enumerate}\setlength\itemsep{1.4mm}
  \item For questions where candidates must write their answers, do not award
        the mark if the answer is incomplete, or if there is additional
        information that is irrelevant or shifts the focus of the answer.
  \item For questions where candidates are asked to answer using words as they
        appear in the text, do not accept paraphrases.
  \item Accept spelling and grammatical mistakes provided they do not change
        the meaning of the answer.
  \item For questions where candidates must choose an answer from options, if
        two answers are given -- one in the box and one outside -- only mark
        the answer inside the box.
  \item Words inside brackets in this markscheme are optional: candidates may
        include these words in their answer, but they are not essential for
        the mark to be awarded. If there is more than one possible answer to a
        question, these answers are indicated by a slash (/).
\end{enumerate}
\endgroup
