\ibaudiotext{A}{You are going to hear an interview between a journalist and the
  coordinator of a city bicycle-sharing scheme.}

\ibrubricmcq
\begin{ibmcq}[A]{The scheme was originally set up in order to\dots}
  \opt{reduce the number of short car journeys.}
  \opt{raise money for the city council.}
  \opt{replace part of the bus network.}
  \opt{attract visitors to the city centre.}
\end{ibmcq}
\begin{ibmcq}[B]{Before riding, users must\dots}
  \opt{book a bicycle the day before.}
  \opt{register once using an application.}
  \opt{pay a fee for each month.}
  \opt{collect a helmet from the station.}
\end{ibmcq}
\begin{ibmcq}[D]{Most of the damage to the bicycles is caused by\dots}
  \opt{theft.}
  \opt{accidents.}
  \opt{bad weather.}
  \opt{ordinary use.}
\end{ibmcq}
\begin{ibmcq}[C]{The coordinator says the biggest surprise has been\dots}
  \opt{how far people ride.}
  \opt{how early people ride.}
  \opt{who uses the bicycles.}
  \opt{how few bicycles are lost.}
\end{ibmcq}
\begin{ibmcq}[B]{The interview is mainly about the\dots}
  \opt{cost of running the scheme.}
  \opt{way the scheme has developed.}
  \opt{safety of cycling in the city.}
  \opt{plans to close the scheme down.}
\end{ibmcq}

\ibrubricmcq
\begin{ibmcq}[A]{The boat club supports the regatta mainly in order to\dots}
  \opt{attract new members.}
  \opt{sell its old equipment.}
  \opt{please its sponsors.}
  \opt{train its own coaches.}
\end{ibmcq}
\begin{ibmcq}[C]{According to the manager, the club's main difficulty is\dots}
  \opt{finding qualified coaches.}
  \opt{booking water for training.}
  \opt{the cost of replacing boats.}
  \opt{competition from other sports.}
\end{ibmcq}
\begin{ibmcq}[B]{What does the manager say puts most beginners off?}
  \opt{The cost of joining.}

  \opt{The lack of changing rooms.}
\end{ibmcq}
\begin{ibmcq}[D]{The club's open day is intended to show that rowing is\dots}
  \opt{safer than people think.}
  \opt{suitable for all ages.}

  \opt{cheaper than people expect.}
\end{ibmcq}

  \opt{Most call-outs happen during the winter months.}
